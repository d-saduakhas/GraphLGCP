Model 1

In this section, we fit Model 1 as described in the paper. This model includes only spatial effects, without any covariates. We show two fitting strategies:

  • lgcp_graph wrapper (our convenient interface, recommended)
  • Manual INLA (standard INLA workflow on the metric graph, for advanced users)

We use a mesh with edge-wise spacing of 50 meters — this is the configuration reported in the paper.

0.1 Data Loading and Graph Preparation

This part loads the required libraries, reads the data and graph files, and builds a mesh using the graph structure. The operations prepare the spatial framework needed for the subsequent modeling steps.

1 Fitting with lgcp_graph wrapper (50 m mesh)

The lgcp_graph wrapper is the recommended way to fit LGCP models on metric graphs. It internally handles the integration weights, the INLA data stack, and the exposure variable E, simplifying the workflow considerably. We only need to add the observed events, provide covariates at mesh locations, and specify the model formula.

graph$compute_mesh_weights()

# Add observations to extract locations
rem.obs <- graph$add_observations(
  data, normalized = TRUE, tolerance = 0.6, clear_obs = TRUE,
  duplicated_strategy = "jitter"
)

# Extract prepared data and count events
ss <- graph$get_data()
n_accidents <- sum(!is.na(ss$AccidentSeverityId))

# Reset observations to contain only the events (wrapper computes exposures, INLA stack, data internally)
graph$clear_observations()

obs_50 <- data.frame(
  y = rep(1, n_accidents),
  edge_number = ss$.edge_number,
  distance_on_edge = ss$.distance_on_edge,
  Intercept = 1,
  .group = 1
)

graph$add_observations(data = obs_50, normalized = TRUE, duplicated_strategy = "jitter")

# Provide a manual covariate that is constant along edges (intercept)
manual_cov_50 <- data.frame(
  Intercept = rep(1, nrow(graph$mesh$VtE)),
  edge_number = graph$mesh$VtE[, 1],
  distance_on_edge = graph$mesh$VtE[, 2],
  .group = 1,
  coord_x = graph$mesh$V[,1],
  coord_y = graph$mesh$V[,2]
)

# Build the SPDE model object (used by the wrapper in the formula)
model_iface_50 <- graph_spde(graph, alpha = 1, parameterization = "spde", 
                             LGCP = TRUE, shared_lib="rSPDE")

# Wrapper fit on the 50 m mesh
system.time(fit_iface_50 <- lgcp_graph(
  y ~ -1 + Intercept + f(field, model = model_iface_50),
  graph = graph,
  manual_covariates = manual_cov_50,
  interpolate = FALSE,
  use_current_mesh = TRUE,
  control.compute = list(config = TRUE),
  control.fixed   = list(prec = list(Intercept = 0)),
  control.inla    = list(control.vb = list(enable = FALSE), int.strategy = "eb")
))

1.1 Visualizing Model Results

When fitting with lgcp_graph, the returned object includes a graph_fitted_data data frame containing predictions at both observation and integration locations. We can plot the three key outputs directly using graph$plot_function:

  • Linear predictor: posterior mean of the linear predictor
  • Predicted field: posterior mean of the latent Gaussian field \(u(s)\)
  • Log fitted values: log of the predicted intensity
library(scico)
library(ggplot2)
library(viridis)

# Posterior mean of the log-intensity (linear predictor)
png("Figures/model1_linear_predictor2.png", height = 3 * 700, width = 3 * 900, res = 300)
p1 <- graph$plot_function(data = ".linear_predictor", newdata = fit_iface_50$graph_fitted_data,
                    vertex_size = 0, edge_width = 0.2, scale_color = scale_color_scico(palette = "vik", midpoint = 0)) + theme_minimal()

2 Alternative: Manual INLA workflow (50 m mesh)

For users who want full control over the INLA stack and data preparation, we show here the equivalent manual workflow. This approach requires explicitly constructing the integration points, building the INLA stack, and passing the exposure variable E to inla().

2.1 Preparing the Spatial Observations

Before fitting the model, we prepare the spatial observations by defining integration points and associating them with the mesh. This segmentation allows the model to differentiate between observed locations and additional integration nodes. The e field holds integration weights.

alpha <- 1
atilde <- graph$mesh$weights

# Clear and add observations explicitly for modeling
graph$clear_observations()

# Add the observed accidents
graph$add_observations(
  data = data.frame(
    y = rep(1, n_accidents),
    e = 0,
    edge_number = ss$.edge_number,
    distance_on_edge = ss$.distance_on_edge,
    Intercept = 1
  ),
  normalized = TRUE
)

# Add integration points
graph$add_observations(
  data = data.frame(
    y = rep(0, length(atilde)),
    e = atilde,
    edge_number = graph$mesh$VtE[, 1],
    distance_on_edge = graph$mesh$VtE[, 2],
    Intercept = 1
  ),
  normalized = TRUE
)

2.2 Creating the spatial model (Manual INLA on 50 m mesh)

We build the SPDE model on the graph, construct the INLA stack, and fit the log-Gaussian Cox process without covariates. We use the “spde” parameterization. (If switching to “matern”, see the commented snippet for converting to SPDE parameters.)

# Create the Graph SPDE model (spde parameterization)
system.time(
  model_manual_50 <- graph_spde(graph, alpha = alpha, parameterization = "spde", verbose = 1)
)
##    user  system elapsed 
##  25.686   0.332  26.112
saveRDS(model_manual_50,  "Data/model1_graph_spde_50m_7FRC.rds", compress = "xz")

# Build the INLA stack for Manual INLA
data_manual_50 <- graph_data_spde(graph_spde = model_manual_50, name = "field", covariates = c("Intercept"))
saveRDS(data_manual_50,"Data/model1_graph_data_spde_50m_7FRC.rds", compress = "xz")

stk_50 <- inla.stack(
  data    = data_manual_50[["data"]],
  A       = data_manual_50[["basis"]],
  effects = data_manual_50[["index"]]
)

# Manual INLA fit on the 50 m mesh
fit_manual_50 <- inla(
  y ~ -1 + Intercept + f(field, model = model_manual_50),
  family = "poisson",
  data = inla.stack.data(stk_50),
  control.predictor = list(A = inla.stack.A(stk_50), compute = TRUE, link = 1),
  control.compute   = list(config = TRUE),  # needed for excursion sets
  control.fixed     = list(prec = list(Intercept = 0)),  # manual intercept - default prior correction
  control.inla      = list(control.vb = list(enable = FALSE), int.strategy = "eb"),
  E = inla.stack.data(stk_50)$e
)
summary(fit_manual_50)
