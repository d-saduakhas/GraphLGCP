Model 2

In this section, we fit Model 2 as described in the paper, which extends Model 1 by incorporating spatial covariates:

\[\log(\lambda(s)) = \beta_0 + \sum_{i=1}^{11} \beta_i X_i(s) + u(s)\]

where \(X_i\) are spatial covariates and \(u(s)\) is the Whittle–Matérn field. The covariates include functional road class (FRC, one-hot encoded), speed limit, and log-transformed distances to amenities (education, finance, hospitals, intersections). Details on covariate extraction are in Extracting Covariates and Computing Spatial Covariates.

We fit the model using the lgcp_graph wrapper from the MetricGraph package.

2 Adding Covariates to the Graph

We add the accident observations to the graph and extract edge-level covariates (FRC, speed limit, road length). The length covariate is converted from meters to kilometers for consistency with the distance-to-amenity covariates.

# Add raw observations to graph
graph$add_observations(
  data = data, normalized = TRUE, tolerance = 0.6, group = "YearNew",
  clear_obs = TRUE, duplicated_strategy = "jitter"
)

# Create a "Length" covariate (edge length in km)
graph$set_edge_weights(weights = graph$get_edge_weights() %>%
                         mutate(Length = graph$get_edge_lengths()))
data_on_graph <- graph$get_data()
data_on_graph$Length <- data_on_graph$Length / 1000  # convert to km

ss <- graph$get_data()
graph$compute_mesh_weights()
n_accidents <- sum(!is.na(ss$AccidentSeverityId))  # 2482

3 Preparing Covariates at Mesh Locations

For the lgcp_graph wrapper, covariates must be available at both observation and mesh locations. We extract edge weights at mesh nodes using edgeweight_to_data() and merge with the precomputed amenity distances.

alpha <- 1
atilde <- graph$mesh$weights

# Get covariates at mesh locations from edge weights
data_on_mesh <- graph$edgeweight_to_data(mesh = TRUE, add = FALSE,
                                          return = TRUE, normalized = TRUE) %>%
  dplyr::filter(.group == 2015)

# Merge with precomputed amenity distances
data_on_mesh <- merge(data_on_mesh, mesh_covariates,
                       by = c(".edge_number", ".distance_on_edge"))
data_on_mesh <- st_as_sf(data_on_mesh, coords = c("x", "y"), crs = 4326)

YearNew <- unique(data_on_graph$YearNew)
n_accidents <- table(data_on_graph$YearNew)

4 Normalizing Covariates

All covariates are normalized using min–max scaling applied jointly across observation and mesh locations to ensure consistency. This is important because the covariates must be on comparable scales for the model to fit well.

min_max_normalize <- function(x) {
  return((x - min(x)) / (max(x) - min(x)))
}

# Columns to normalize
columns_to_normalize <- c("Length", "SpeedLimit",
                          "log_distance_to_finance", "log_distance_to_hospital",
                          "log_distance_to_intersection", "log_distance_to_education")

# Combine graph and mesh data, normalize jointly, then split back
graph_data <- data_on_graph[, columns_to_normalize]
mesh_data  <- data_on_mesh[, columns_to_normalize]
combined   <- dplyr::bind_rows(graph_data, mesh_data) %>% dplyr::select(-geometry)

for (col in columns_to_normalize) {
  combined[[col]] <- min_max_normalize(combined[[col]])
}

normalized_graph_data <- combined[1:nrow(graph_data), ]
normalized_mesh_data  <- combined[(nrow(graph_data) + 1):nrow(combined), ]
data_on_graph[, columns_to_normalize] <- normalized_graph_data
data_on_mesh[, columns_to_normalize] <- normalized_mesh_data

# Reorder mesh data to match graph$mesh$VtE ordering
data_on_mesh <- match_mesh_data(graph, data_on_mesh)
# Verify ordering:
sum((graph$mesh$VtE - as.matrix(sf::st_drop_geometry(data_on_mesh[, 1:2])))^2)

8 Fitting Model 2

We build the SPDE model and fit using the lgcp_graph wrapper. The formula includes the intercept, speed limit, FRC categories (1–6, with FRC7 as reference), log-distances to amenities, and the Whittle–Matérn field.

spde_model <- graph_spde(graph, alpha = 1, LGCP = TRUE, parameterization = "spde")

system.time(fit_model2 <- lgcp_graph(
  y ~ -1 + Intercept + speed_limit +
    road_type1 + road_type2 + road_type3 + road_type4 + road_type5 + road_type6 +
    log_dist_education + log_dist_hospital + log_dist_finance + log_dist_intersection +
    f(field, model = spde_model),
  graph = graph,
  manual_covariates = manual_cov,
  interpolate = FALSE,
  use_current_mesh = TRUE,
  control.compute = list(config = TRUE, waic = TRUE, dic = TRUE, cpo = TRUE),
  control.fixed = list(prec = list(Intercept = 0, speed_limit = 0,
                                    road_type1 = 0, road_type2 = 0, road_type3 = 0,
                                    road_type4 = 0, road_type5 = 0, road_type6 = 0)),
  control.inla = list(control.vb = list(enable = FALSE), int.strategy = "eb")
))
